Ejemplo de Operaciones SQL

-- CREATE
INSERT INTO tblPeliculas
(Titulo, DuracionMinutos, PrecioBase, ClasificacionEdad, Genero)
VALUES
('Avatar 3', 181, 15000, 'PG', 'Acción');

-- READ
SELECT *
FROM tblPeliculas;

-- UPDATE
UPDATE tblPeliculas
SET PrecioBase = 18000
WHERE PeliculaId = 1;

-- DELETE
DELETE FROM tblPeliculas
WHERE PeliculaId = 3;
